{{- /* Dati del mosaico di una sezione: usati sia dal mosaico (mosaico.html) sia dai dati strutturati ItemList (head.html), così le due cose contengono sempre le stesse opere nello stesso ordine. Restituisce: dict "items" (lista ordinata) Ogni item: page, w, h (cm della tela, per l'etichetta), img (risorsa), r (proporzione larghezza/altezza della foto), disponibile, nome, etichetta ORDINE: lo stesso dell'elenco sotto il mosaico (dalla più recente alla più vecchia), così miniature ed elenco si leggono insieme. Le miniature NON sono in scala: hanno tutte la stessa altezza (vedi .mosaico-tessera in assets/css/override.css) e la foto è mostrata intera. */ -}} {{- $items := slice -}} {{- range where site.RegularPages "Section" .Section -}} {{- /* stesso elenco e stesso ordine di partials/post_list.html */ -}} {{- $p := . -}} {{- $w := 0.0 -}} {{- $h := 0.0 -}} {{- with $p.Params.schema -}} {{- $w = partial "numero-cm.html" .width -}} {{- $h = partial "numero-cm.html" .height -}} {{- end -}} {{- $img := "" -}} {{- /* "miniatura_mosaico" (facoltativa): foto ritagliata solo per il mosaico, utile quando la foto principale ha molto sfondo intorno */ -}} {{- with $p.Params.miniatura_mosaico | default $p.Params.image -}}{{- $img = resources.Get (strings.TrimPrefix "/" .) -}}{{- end -}} {{- if not $img -}} {{- warnf "[mosaico] %s escluso: immagine (image) mancante o non trovata" $p.File.Path -}} {{- continue -}} {{- end -}} {{- /* disponibilità: stessa regola di partials/vendita.html (vendita ha la precedenza) */ -}} {{- $disp := false -}} {{- with $p.Params.vendita -}} {{- $disp = eq (.stato | default "disponibile") "disponibile" -}} {{- else -}} {{- $disp = $p.Params.disponibile | default false -}} {{- end -}} {{- /* etichetta: nome, tecnica, misure, anno (solo le parti presenti) */ -}} {{- $nome := $p.Params.schema.name | default $p.Params.heading | default (index (split $p.Title " — ") 0) -}} {{- $nome = trim (replace $nome "*" "") " " -}} {{- /* l'asterisco degli studi non serve nell'etichetta */ -}} {{- $parti := slice $nome -}} {{- with $p.Params.schema.medium -}}{{- $parti = $parti | append (lower .) -}}{{- end -}} {{- if and (gt $w 0.0) (gt $h 0.0) -}} {{- $parti = $parti | append (replace (printf "%g×%g cm" $w $h) "." ",") -}} {{- end -}} {{- with $p.Params.schema.dateCreated -}}{{- $parti = $parti | append (printf "%v" .) -}}{{- end -}} {{- if $disp -}}{{- $parti = $parti | append "disponibile" -}}{{- end -}} {{- $etichetta := delimit $parti ", " -}} {{- $r := div (float $img.Width) (float $img.Height) -}} {{- $items = $items | append (dict "page" $p "w" $w "h" $h "img" $img "r" $r "disponibile" $disp "nome" $nome "etichetta" $etichetta) -}} {{- end -}} {{- return (dict "items" $items) -}}
